--14. view geo_map
DROP VIEW IF EXISTS tmp.v_geo_map;
CREATE VIEW tmp.v_geo_map AS
SELECT s.id as site_id, coalesce(s.name, 'dump') as site, 
-- s.subregion_id as subregion_id, 
coalesce(sr.name, 'dump') as subregion,
-- sr.region_id as region_id, 
coalesce(r.name, 'dump') as region,
-- r.country_id as country_id, 
c.name as country
FROM tmp.sites_map as s
JOIN tmp.subregions_map as sr
ON s.subregion_id = sr.id
JOIN tmp.regions_map as r
ON sr.region_id = r.id
JOIN tmp.countries_map as c
ON r.country_id = c.id
