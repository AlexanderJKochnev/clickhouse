--winery
DROP VIEW IF EXISTS tmp.v_wineries_map;
CREATE VIEW tmp.v_wineries_map AS
SELECT p.id AS id, l.winery AS name
FROM (SELECT DISTINCT winery FROM tmp.flat_drinks_lake) AS l
JOIN postgresql('pg_dev:5432', 'wine_db', 'producers', 'wine', 'wine1') AS p
ON unaccent(l.winery) = unaccent(p.name);

--GEO
--countries
DROP VIEW IF EXISTS tmp.v_countries_map;
CREATE VIEW tmp.v_countries_map AS
SELECT DISTINCT
    pc.id AS id,
    fdl.country AS name
FROM tmp.flat_drinks_lake AS fdl
LEFT JOIN postgresql('pg_dev:5432', 'wine_db', 'countries', 'wine', 'wine1') AS pc 
    ON pc.name = transform(fdl.country, 
        ['US', 'Republic of Georgia', 'Republic of Macedonia', 'Russia', 'Vietnamese'], 
        ['United States', 'Georgia', 'North Macedonia', 'Russian Federation', 'Vietnam'], 
        fdl.country
    )
WHERE id IS NOT NULL
ORDER BY id, name;

--regions
DROP VIEW IF EXISTS tmp.v_regions_map;
CREATE VIEW tmp.v_regions_map AS
WITH c1 AS (
SELECT DISTINCT country_id, name, reg FROM (
SELECT p.id AS country_id, 
l.region AS name, l.reg AS reg
FROM (SELECT DISTINCT country, coalesce(region, 'dump') AS reg, region FROM tmp.flat_drinks_lake) AS l
JOIN tmp.v_countries_map AS p
ON l.country = p.name
)), 
SELECT p.id AS id, name, country_id
FROM c1
JOIN postgresql('pg_dev:5432', 'wine_db', 'regions', 'wine', 'wine1') AS p
ON unaccent(reg) = unaccent(coalesce(p.name,'dump')) 
AND c1.country_id = p.country_id
WHERE country_id IS NOT NULL
ORDER BY id, name, country_id;

--subregions
DROP VIEW IF EXISTS tmp.v_subregions_map;
CREATE VIEW tmp.v_subregions_map AS
WITH c1 AS (
SELECT DISTINCT country_id, reg, name, sub FROM (
SELECT p.id AS country_id, 
l.reg AS reg,
l.subregion AS name, l.sub AS sub
FROM (SELECT DISTINCT country, 
                      coalesce(region, 'dump') AS reg,
                      coalesce(subregion, 'dump') AS sub,
                      subregion
      FROM tmp.flat_drinks_lake WHERE country IS NOT NULL) AS l
JOIN tmp.v_countries_map AS p
ON l.country = p.name
)), c2 AS (
SELECT  
p.id AS region_id, c1.name as name, c1.sub
FROM c1
JOIN tmp.v_regions_map AS p
ON reg = coalesce(p.name,'dump')  
AND c1.country_id = p.country_id
) 
SELECT sr.id AS id, 
c2.name AS name, c2.region_id AS region_id
FROM c2
LEFT JOIN postgresql('pg_dev:5432', 'wine_db', 'subregions', 'wine', 'wine1') AS sr
ON unaccent(c2.sub) = unaccent(coalesce(sr.name, 'dump'))
AND c2.region_id = sr.region_id
ORDER BY id, name, region_id;

--sites
DROP VIEW IF EXISTS tmp.v_sites_map;
CREATE VIEW tmp.v_sites_map AS
WITH c1 AS (
SELECT DISTINCT country_id, reg, sub, site, name FROM (
SELECT p.id AS country_id, 
l.reg AS reg,
l.sub AS sub,
l.dsite AS site,
l.name AS name
FROM (SELECT DISTINCT country, 
                      coalesce(region, 'dump') AS reg,
                      coalesce(subregion, 'dump') AS sub,
                      site AS name,
                      coalesce(site, 'dump') AS dsite
      FROM tmp.flat_drinks_lake WHERE country IS NOT NULL) AS l
JOIN tmp.v_countries_map AS p
ON l.country = p.name
)), c2 AS (
SELECT  
p.id AS region_id, 
c1.name as name,
c1.site as site,
c1.sub
FROM c1
JOIN tmp.v_regions_map AS p
ON reg = coalesce(p.name,'dump')  
AND c1.country_id = p.country_id
), c3 AS ( 
SELECT sr.id AS subregion_id, 
c2.name AS name,
c2.region_id AS region_id,
c2.site as site,
c2.sub
FROM c2
JOIN postgresql('pg_dev:5432', 'wine_db', 'subregions', 'wine', 'wine1') AS sr
ON unaccent(c2.sub) = unaccent(coalesce(sr.name, 'dump'))
AND c2.region_id = sr.region_id
ORDER BY id, name, region_id
)
SELECT 
si.id as id,
c3.name as name, 
c3.subregion_id as subregion_id
FROM c3
JOIN postgresql('pg_dev:5432', 'wine_db', 'sites', 'wine', 'wine1') AS si
ON unaccent(c3.site) = unaccent(coalesce(si.name, 'dump'))
AND c3.subregion_id = si.subregion_id
;

--MANY TO MANY VIEWS
-- 1. Таблица связей для сортов винограда (varietals)
CREATE OR REPLACE VIEW tmp.v_drinks_varietals AS
SELECT 
    lake.id AS drink_id,
    v.id AS varietal_id
FROM (SELECT id, arrayJoin(varietals) AS name FROM tmp.flat_drinks_lake) AS lake
INNER JOIN tmp.v_varietal_map AS v ON lake.name = v.name
ORDER BY drink_id, varietal_id
;

-- 2. Таблица связей для блюд (foods)
CREATE OR REPLACE VIEW tmp.v_drinks_foods AS
SELECT 
    lake.id AS drink_id,
    f.id AS food_id
FROM (SELECT id, arrayJoin(foods) AS name FROM tmp.flat_drinks_lake) AS lake
INNER JOIN tmp.v_foods_map AS f ON lake.name = f.name
ORDER BY drink_id, food_id
;

-- 3. Таблица связей для дегустационных заметок (tasting_notes)
CREATE OR REPLACE VIEW tmp.v_drinks_tasting_notes AS
SELECT 
    lake.id AS drink_id,
    t.id AS tasting_note_id
FROM (SELECT id, arrayJoin(tasting_notes) AS name FROM tmp.flat_drinks_lake) AS lake
INNER JOIN tmp.v_tasting_notes_map AS t ON lake.name = t.name
ORDER BY drink_id, tasting_note_id  
;

-- 4. Таблица связей для базовых ингредиентов (base_ingredient)

CREATE OR REPLACE VIEW tmp.v_drinks_base_ingredients AS
SELECT 
    lake.id AS drink_id,
    b.id AS base_ingredient_id
FROM (SELECT id, arrayJoin(base_ingredient) AS name FROM tmp.flat_drinks_lake) AS lake
INNER JOIN tmp.v_base_ingredients_map AS b ON lake.name = b.name
ORDER BY drink_id, base_ingredient_id;
