CREATE OR REPLACE TABLE tmp.subcat AS
WITH o AS (
SELECT  lake.id,
        sub.id as subcategory_id,
        lake.beer_type,
        sub.name
FROM tmp.flat_drinks_lake AS lake
INNER JOIN tmp.pg_subcategories AS sub 
ON hasAny(lake.beer_type, sub.synonims)
), one AS (
SELECT o.id
FROM o
GROUP BY o.id
HAVING COUNT() = 1
)
SELECT DISTINCT ON (o.id) o.id as id, o.subcategory_id as subcategory_id
FROM o 
WHERE o.id IN (SELECT id FROM one)
ORDER BY id, subcategory_id;
