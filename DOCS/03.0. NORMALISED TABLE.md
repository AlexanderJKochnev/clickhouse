INSERT INTO tmp.flat_drinks_normalized
(
ch_id, title, description, subcategory_id, source_id, producer_id, parcel_id, site_id, glassware_id,
scale_id, body_id
)
SELECT  
lake.id as ch_id, 
lake.wine as title, 
lake.review as description,
sub.subcategory_id as subcategory_id,
so.id as source_id,
win.id as producer_id,
nullif(des.id, 0) as parcel_id, 
geo.site_id as site_id,
nullif(glass.id, 0) as glassware_id, 
nullif(sca.id, 0) as scale_id,
nullif(bod.id,0) as body_id
-- , lake.winery, lake.site, lake.subregion, lake.region, lake.country 

FROM tmp.flat_drinks_lake AS lake
LEFT JOIN tmp.v_designations_map AS des ON lake.designation = des.name
LEFT JOIN tmp.wineries_map as win ON lake.winery = win.name
LEFT JOIN tmp.v_geo_map as geo
ON  coalesce(lake.site, 'dump') = geo.site
AND coalesce(lake.subregion, 'dump') = geo.subregion 
AND coalesce(lake.region, 'dump') = geo.region
AND lake.country = geo.country
LEFT JOIN tmp.v_glasswares_map as glass ON unaccent(lake.glassware) = unaccent(glass.name)
LEFT JOIN tmp.v_scales_map as sca ON unaccent(lake.scale) = unaccent(sca.name) 
LEFT JOIN tmp.v_bodies_map as bod ON unaccent(lake.body) = unaccent(bod.name)
LEFT JOIN tmp.pg_source as so ON lake.source = so.name
LEFT JOIN tmp.subcat as sub ON lake.id = sub.id
