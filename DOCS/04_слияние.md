## создаем таблицы с индексированными массивами 
CREATE TABLE tmp.flat_drinks_indexed
(
    ch_id UInt64,
    title_arr Array(String),
    -- Создаем Блум-фильтр для токенов массива. 
    -- 30720 — размер фильтра в байтах, 2 — количество хэш-функций (оптимально для строк)
    INDEX arr_idx title_arr TYPE tokenbf_v1(30720, 2, 0) GRANULARITY 1
) ENGINE = MergeTree()
ORDER BY ch_id;

-- Таблица 2
CREATE TABLE tmp.pg_drinks_indexed
(
    id UInt64,
    title_arr Array(String),
    INDEX arr_idx title_arr TYPE tokenbf_v1(30720, 2, 0) GRANULARITY 1
) ENGINE = MergeTree()
ORDER BY id;

) ENGINE = MergeTree()

## сравнение
### полное совпадение DONE
INSERT INTO tmp.drinks_ch_pg
(id, ch_id)
SELECT DISTINCT pg.id, ch.ch_id
FROM tmp.flat_drinks_indexed AS ch
JOIN tmp.pg_drinks_indexed AS pg
ON arraySort(ch.title_arr) = arraySort(pg.title_arr)

### слияние 
INSERT INTO FUNCTION postgresql('pg_dev:5432', 'wine_db', 'drink_update', 'wine', 'wine1')
(id, ch_id, title, description)
SELECT d.id, n.ch_id, n.title, n.description
FROM tmp.flat_drinks_normalized AS n
JOIN tmp.drinks_ch_pg AS d
ON n.ch_id = d.ch_id
LIMIT 10\G

### обновление 
docker compose exec -i wine_host psql -U wine -d wine_db -c "
UPDATE drinks
SET ch_id = du.ch_id,
    description = du.description
FROM drink_update AS du
WHERE drinks.id = du.id
;
"

INSERT INTO FUNCTION postgresql('pg_dev:5432', 'wine_db', 'drink_insert', 'wine', 'wine1')
(ch_id, title, description, subcategory_id, source_id, producer_id, parcel_id, site_id, 
glassware_id, scale_id, body_id)
SELECT DISTINCT ON (title) ch.ch_id, ch.title, ch.description, ch.subcategory_id, ch.source_id, 
nullif(producer_id, 0) AS producer_id, 
parcel_id, 
nullif(site_id, 0) AS site_id,
glassware_id, scale_id, body_id
FROM tmp.flat_drinks_normalized AS ch
LEFT JOIN postgresql('pg_dev:5432', 'wine_db', 'drinks', 'wine', 'wine1') AS pg
ON ch.ch_id = pg.ch_id
WHERE pg.ch_id IS NULL
AND ch.site_id != 0
AND subcategory_id != 0;
